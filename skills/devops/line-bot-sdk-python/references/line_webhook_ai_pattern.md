# LINE Webhook with AI Integration (UX Pattern)\n\nWhen building LINE bots that rely on slow LLM processing (taking more than 1-2 seconds), the webhook handler MUST return HTTP 200 OK immediately to prevent LINE from timing out and retrying the message.\n\n## The `showLoadingAnimation` Pattern\nTo provide the best UX, use LINE's loading animation endpoint. This displays a bouncing dot animation in the chat while the background task processes the AI request.\n\n```python\nfrom fastapi import FastAPI, Request, Header, HTTPException, status, BackgroundTasks\nfrom typing import Annotated\nfrom linebot.v3.webhook import WebhookHandler\nfrom linebot.v3.exceptions import InvalidSignatureError\nfrom linebot.v3.messaging import Configuration, ApiClient, MessagingApi, ReplyMessageRequest, ShowLoadingAnimationRequest, FlexMessage, FlexContainer\n\nhandler = WebhookHandler(\"YOUR_SECRET\")\nconfiguration = Configuration(access_token=\"YOUR_TOKEN\")\n\n@app.post(\"/webhook/line\")\nasync def line_webhook(\n    request: Request,\n    background_tasks: BackgroundTasks,\n    x_line_signature: Annotated[str | None, Header(alias=\"X-Line-Signature\")] = None,\n):\n    body = await request.body()\n    body_str = body.decode(\"utf-8\")\n    try:\n        handler.handle(body_str, x_line_signature)\n    except InvalidSignatureError:\n        raise HTTPException(status_code=401)\n\n    # Note: If the official handler routes messages before returning, it blocks.\n    # To return immediately, parse JSON directly, dispatch to background, and return 200.\n    import json\n    data = json.loads(body_str)\n    for event in data.get(\"events\", []):\n        if event.get(\"type\") == \"message\":\n            background_tasks.add_task(process_ai_task, event)\n\n    return {\"status\": \"ok\"}\n\nasync def process_ai_task(event: dict):\n    user_id = event[\"source\"][\"userId\"]\n    reply_token = event[\"replyToken\"]\n    \n    with ApiClient(configuration) as api_client:\n        line_bot_api = MessagingApi(api_client)\n        \n        # 1. Start loading animation (UX)\n        try:\n            line_bot_api.show_loading_animation(ShowLoadingAnimationRequest(chatId=user_id, loadingSeconds=30))\n        except Exception:\n            pass\n        \n        # 2. Run Heavy AI Task\n        import asyncio\n        await asyncio.sleep(5) \n        \n        # 3. Reply with Flex Message\n        flex_dict = { \"type\": \"bubble\", \"body\": { \"type\": \"box\", \"layout\": \"vertical\", \"contents\": [ { \"type\": \"text\", \"text\": \"✅ Processed\" } ] } }\n        line_bot_api.reply_message(ReplyMessageRequest(\n            reply_token=reply_token,\n            messages=[FlexMessage(alt_text=\"Result\", contents=FlexContainer.from_dict(flex_dict))]\n        ))\n```
